PROBLEM 1: Salary Calculation
gross = basic + houseRent + medical - tax
============================================================

INCLUDE Irvine32.inc

.data
basic     DWORD 50000
houseRent DWORD 15000
medical   DWORD 8000
tax       DWORD 7000
gross     DWORD ?
msg       BYTE "Gross Salary: Rs. ", 0

.code
main PROC
    mov eax, basic
    add eax, houseRent
    add eax, medical
    sub eax, tax
    mov gross, eax

    mov edx, OFFSET msg
    call WriteString
    mov eax, gross
    call WriteDec
    call Crlf

    call DumpRegs
    exit
main ENDP
END main

; Gross Salary: Rs. 66000
; (EAX = 00010230h = 66000)

PROBLEM 2: Array Sum and Average
DWORD arr = 10,20,30,40,50
============================================================

INCLUDE Irvine32.inc

.data
arr     DWORD 10, 20, 30, 40, 50
sum     DWORD ?
average DWORD ?
msg1    BYTE "Sum: ", 0
msg2    BYTE "Average: ", 0

.code
main PROC
    mov esi, OFFSET arr
    mov ecx, LENGTHOF arr
    mov ebx, 0

SumLoop:
    add ebx, [esi]
    add esi, TYPE arr
    loop SumLoop

    mov sum, ebx

    mov eax, ebx
    mov edx, 0
    mov ecx, LENGTHOF arr
    div ecx
    mov average, eax

    mov edx, OFFSET msg1
    call WriteString
    mov eax, sum
    call WriteDec
    call Crlf

    mov edx, OFFSET msg2
    call WriteString
    mov eax, average
    call WriteDec
    call Crlf

    exit
main ENDP
END main

; EXPECTED OUTPUT:
; Sum: 150
; Average: 30

PROBLEM 4: Find Maximum Element
WORD arr = 45, 12, 78, 33, 90
============================================================

INCLUDE Irvine32.inc

.data
arr  WORD 45, 12, 78, 33, 90
max  WORD ?
msg  BYTE "Maximum: ", 0

.code
main PROC
    mov esi, OFFSET arr
    mov ecx, LENGTHOF arr
    mov ax, [esi]
    add esi, TYPE arr
    dec ecx

FindMax:
    mov bx, [esi]
    cmp bx, ax
    jle Skip
    mov ax, bx
Skip:
    add esi, TYPE arr
    loop FindMax

    mov max, ax

    mov edx, OFFSET msg
    call WriteString
    movzx eax, max
    call WriteDec
    call Crlf

    exit
main ENDP
END main

; EXPECTED OUTPUT:
; Maximum: 90
